JS · 按压反馈与完成态（五按钮共用）

// components/kawaii-btn/index.js
Component({
  properties: {
    label: String, icon: String,
    variant: { type: String, value: 'plain' }, // plain|primary|sun
    doneText: { type: String, value: '' }
  },
  data: { pressed: false, done: false },
  methods: {
    onTap() {
      if (this.data.done) return;
      this.setData({ pressed: true });
      wx.vibrateShort({ type: 'light' });
      setTimeout(() => this.setData({ pressed: false }), 120);
      this.triggerEvent('act', {
        finish: () => {
          this.setData({ done: true });      // 变完成态+盖章
          setTimeout(() => this.setData({ done: false }), 2000);
        }
      });
    }
  }
});
